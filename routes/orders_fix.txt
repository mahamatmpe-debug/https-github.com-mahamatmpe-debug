async function generatePdf(data) {
  const pdf = await PDFDocument.create();
  const page = pdf.addPage([600, 800]);
  const font = await pdf.embedFont(StandardFonts.Helvetica);
  let y = 760;
  page.drawText('BON DE COMMANDE - CNFFD 2026', { x: 40, y: y, size: 16, font, color: rgb(0,0,0.5) });
  y -= 30;
  page.drawText('ID: ' + data.id + ' | Date: ' + new Date(data.date).toLocaleString('fr-FR'), { x: 40, y: y, size: 10, font });
  y -= 25;
  page.drawText('CLIENT', { x: 40, y: y, size: 12, font });
  y -= 15;
  page.drawText('Nom: ' + (data.customer.name || ''), { x: 50, y: y, size: 10, font });
  page.drawText('Tel: ' + (data.customer.phone || ''), { x: 50, y: y-15, size: 10, font });
  page.drawText('Ville: ' + (data.customer.city || ''), { x: 50, y: y-30, size: 10, font });
  page.drawText('Adr: ' + (data.customer.address || ''), { x: 50, y: y-45, size: 10, font });
  y = y-70;
  page.drawText('LIVRAISON / RETRAIT', { x: 40, y: y, size: 12, font });
  y -= 15;
  page.drawText('Type: ' + (data.delivery.type === 'delivery' ? 'Livraison' : 'Retrait sur place'), { x: 50, y: y, size: 10, font });
  if (data.delivery.type === 'delivery') {
    page.drawText('Frais: ' + data.delivery.fee + ' ' + config.currency, { x: 50, y: y-15, size: 10, font });
    page.drawText('Ville: ' + (data.delivery.city || ''), { x: 50, y: y-30, size: 10, font });
  }
  y = y-50;
  page.drawText('COMPOSANTS', { x: 40, y: y, size: 12, font });
  y -= 15;
  data.cart.forEach((it) => {
    const line = it.name + ' x' + it.qty + ' - ' + it.price + ' ' + config.currency + ' = ' + (it.price*it.qty) + ' ' + config.currency;
    page.drawText(line, { x: 50, y: y, size: 9, font });
    y -= 12;
  });
  y -= 10;
  page.drawText('TOTAL GENERAL: ' + data.total + ' ' + config.currency, { x: 40, y: y, size: 11, font, color: rgb(0.2,0.2,0.8) });
  const pdfBytes = await pdf.save();
  return Buffer.from(pdfBytes);
}
